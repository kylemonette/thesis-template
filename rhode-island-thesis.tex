%% rhode-island-thesis.tex — University of Rhode Island Thesis Template
%%
%% -------------------------------------------------------------------------
%% COMPILE SEQUENCE
%% -------------------------------------------------------------------------
%%
%% OVERLEAF OR LATEXMK (recommended) — nothing extra to do:
%%
%%   Overleaf, and `latexmk -pdf rhode-island-thesis` on your own machine, find every
%%   .aux file that needs BibTeX (including the per-chapter ones and
%%   rhode-island-thesisbib.aux) and rerun pdflatex as needed, in every mode.
%%   genbib.sh is NOT needed.
%%
%% MANUAL COMMAND LINE (offline, without latexmk):
%%
%%   pdflatex rhode-island-thesis
%%   sh genbib.sh
%%
%%   The class writes genbib.sh on every pdflatex run.  It runs bibtex on
%%   each .aux file that needs it (rhode-island-thesis.aux for oneref, one per chapter
%%   for chapterref, plus rhode-island-thesisbib.aux with [bibliography]), then runs
%%   pdflatex twice more.  genbib.txt lists only the bibtex commands, for
%%   systems without sh (e.g. Windows).  If genbib.sh does not yet exist
%%   (very first run), run pdflatex first.
%%
%% -------------------------------------------------------------------------
%% QUICK DRAFT (skip bibliography machinery while writing):
%%
%%   Use the [nobib] option:  \documentclass[nobib,...]{rhode-island-thesis}
%%
%% -------------------------------------------------------------------------
%% CLASS OPTIONS — choose what applies to your thesis
%% -------------------------------------------------------------------------
%%
%%  Degree:       masters (default) | phd
%%  Font size:    12pt (default) | 11pt | 10pt
%%  References:   oneref (default, single list) | chapterref (per-chapter lists)
%%  Format:       standard (default) | manuscript
%%  Committee:    3committee (default) | 4committee | 5committee
%%  Numbering:    sequential (default) | nonsequential
%%  Page numbers: bottomnum (default) | topnum
%%  Citations:    numeric (default) | aparefs
%%  Approval:     hardcopy (default, with sig lines) | electronic
%%  Bibliography: bibliography | nobibliography — a combined alphabetical
%%                BIBLIOGRAPHY section at the end, after any appendices.
%%                Default: on for standard (required by the Graduate School),
%%                off for manuscript.  Use \nocite{*} to list works consulted.
%%  Drafting:     draft | draftbox | simpleref | noprelim | nobib
%%                (see the header of rhode-island-thesis.cls; not for final submission)
%%
%% Example for a PhD dissertation with four committee members:
%%   \documentclass[phd,4committee,topnum]{rhode-island-thesis}

\documentclass[phd,12pt,3committee,chapterref,manuscript]{rhode-island-thesis}

% -------------------------------------------------------------------------
% Packages — add your own here
% -------------------------------------------------------------------------
\usepackage{graphicx}        % \includegraphics
\usepackage{amsmath}         % align, equation*, etc.
\usepackage{amssymb}         % \mathbb, \mathcal, etc.
\usepackage{booktabs}        % \toprule, \midrule, \bottomrule in tables
\usepackage{microtype}       % improved line breaking
\usepackage{hyperref}

% -------------------------------------------------------------------------
% Thesis metadata — fill in every field
% -------------------------------------------------------------------------

\title{The Thesis Title}

\author{John Smith}

\dept{Mathematics}

\copyrightyear{\the\year}

% File that holds your BibTeX entries (without the .bib extension).
% The same file is used for both the per-chapter references and the
% alphabetical bibliography.
\reffile{references}


% -------------------------------------------------------------------------
% Preliminary-page content — each argument is a filename (no extension)
% that will be \input'd.  Comment out any section you don't need.
% Dedication, Preface, and Acknowledgments are all optional.
% Abstract is required.
% -------------------------------------------------------------------------

\abstract{abstract}
\acknowledgements{acknowledgements}
\dedication{dedication}
% Remove if you don't have a preface
\preface{preface}

% -------------------------------------------------------------------------
% Approval page — list the committee members who sign.
% The first \signature is the major professor (advisor).
% Add one \signature per additional member; maximum five total, and the
% count must match the 3committee/4committee/5committee class option.
% -------------------------------------------------------------------------

\signature{Dr.~Jane Smith}          % major professor (signs first)
\signature{Dr.~Robert Johnson}      % committee member 2
\signature{Dr.~Maria Garcia}        % committee member 3
% \signature{Dr.~Fourth Member}     % uncomment for 4committee option
% \signature{Dr.~Fifth Member}      % uncomment for 5committee option

% Name of the current Dean of the Graduate School (the title line
% "DEAN OF THE GRADUATE SCHOOL" is added automatically below it).
\deansignature{Firstname Lastname}

% =========================================================================
\begin{document}
% =========================================================================

% \newchapter{filename} includes a chapter file and triggers the
% preliminary pages (title, approval, abstract, TOC, etc.) on first call.
% Chapter files must be separate .tex files — do not use \chapter directly
% in this root file.

\newchapter{chapter1}
\newchapter{chapter2}
\newchapter{chapter3}

% -------------------------------------------------------------------------
% Appendices (optional) — comment out if you have none.
% Each \newappendix call includes one appendix file.
% -------------------------------------------------------------------------

\newappendix{appendixA}
% \newappendix{appendixB} % uncomment if you have a second appendix

% =========================================================================
\end{document}
% =========================================================================
